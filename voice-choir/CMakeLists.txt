cmake_minimum_required(VERSION 3.24)
project(NooriVoiceChoir VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(CHOIR_BUILD_PLUGIN "Build VST3 and processor tests" ON)
include(CTest)
add_executable(choir_engine_tests tests/EngineTests.cpp)
target_include_directories(choir_engine_tests PRIVATE Source)
add_test(NAME ChoirEngine COMMAND choir_engine_tests)
if(CHOIR_BUILD_PLUGIN)
 if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 deliverable only. Use -DCHOIR_BUILD_PLUGIN=OFF for DSP tests.")
 endif()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
 FetchContent_MakeAvailable(JUCE)
 juce_add_plugin(NooriVoiceChoir COMPANY_NAME "Noori Turab" BUNDLE_ID "audio.nooriturab.voicechoir" PLUGIN_MANUFACTURER_CODE Ntur PLUGIN_CODE Nvch PRODUCT_NAME "NOORI Voice Choir" FORMATS VST3 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Modulation)
 target_sources(NooriVoiceChoir PRIVATE Source/Plugin.cpp Source/Editor.cpp)
 target_compile_definitions(NooriVoiceChoir PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_compile_options(NooriVoiceChoir PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 target_link_libraries(NooriVoiceChoir PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
 add_executable(choir_plugin_tests tests/PluginTests.cpp)
 target_include_directories(choir_plugin_tests PRIVATE Source "${juce_SOURCE_DIR}/modules" $<TARGET_PROPERTY:NooriVoiceChoir,INCLUDE_DIRECTORIES>)
 target_compile_definitions(choir_plugin_tests PRIVATE $<TARGET_PROPERTY:NooriVoiceChoir,COMPILE_DEFINITIONS>)
 target_link_libraries(choir_plugin_tests PRIVATE NooriVoiceChoir)
 add_test(NAME ChoirProcessorAndEditor COMMAND choir_plugin_tests "${CMAKE_BINARY_DIR}")
 set_tests_properties(ChoirProcessorAndEditor PROPERTIES TIMEOUT 120)
endif()
