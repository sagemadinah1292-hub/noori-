cmake_minimum_required(VERSION 3.24)
project(NooriVoiceChoir VERSION 0.4.5 LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
# This file contains public trust configuration only; the signing key stays on Hostinger.
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/Config/Activation.json" NOORI_ACTIVATION_JSON)
string(JSON NOORI_DEFAULT_ACTIVATION GET "${NOORI_ACTIVATION_JSON}" require_activation)
string(JSON NOORI_DEFAULT_SERVER GET "${NOORI_ACTIVATION_JSON}" api_base)
string(JSON NOORI_DEFAULT_PUBLIC_KEY GET "${NOORI_ACTIVATION_JSON}" public_modulus_hex)
option(NOORI_ENABLE_LICENSING "Require signed online registration" ${NOORI_DEFAULT_ACTIVATION})
set(NOORI_LICENSE_SERVER_URL "${NOORI_DEFAULT_SERVER}" CACHE STRING "HTTPS activation API base URL")
set(NOORI_LICENSE_PUBLIC_KEY_HEX "${NOORI_DEFAULT_PUBLIC_KEY}" CACHE STRING "RSA-2048 public modulus, hexadecimal")
set(NOORI_LICENSE_ENABLED_VALUE 0)
if(NOORI_ENABLE_LICENSING)
 if(NOT NOORI_LICENSE_SERVER_URL MATCHES "^https://[^ /]+")
  message(FATAL_ERROR "Protected builds require NOORI_LICENSE_SERVER_URL using HTTPS.")
 endif()
 string(LENGTH "${NOORI_LICENSE_PUBLIC_KEY_HEX}" NOORI_PUBLIC_KEY_LENGTH)
 if(NOT NOORI_PUBLIC_KEY_LENGTH EQUAL 512 OR NOORI_LICENSE_PUBLIC_KEY_HEX MATCHES "[^0-9a-fA-F]")
  message(FATAL_ERROR "Protected builds require the activation server's RSA-2048 public modulus.")
 endif()
 set(NOORI_LICENSE_ENABLED_VALUE 1)
endif()
configure_file(Source/NooriLicenseConfig.h.in generated/NooriLicenseConfig.h @ONLY)
option(CHOIR_BUILD_PLUGIN "Build VST3 and processor tests" ON)
include(CTest)
add_executable(choir_engine_tests tests/EngineTests.cpp)
target_include_directories(choir_engine_tests PRIVATE Source)
add_test(NAME ChoirEngine COMMAND choir_engine_tests)
add_executable(choir_offline_tests tests/OfflineTests.cpp)
target_include_directories(choir_offline_tests PRIVATE Source)
add_test(NAME ChoirOfflineEngine COMMAND choir_offline_tests)
add_executable(choir_musical_tests tests/MusicalTests.cpp)
target_include_directories(choir_musical_tests PRIVATE Source ThirdParty)
add_test(NAME ChoirMusicalParts COMMAND choir_musical_tests)
if(CHOIR_BUILD_PLUGIN)
 if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 deliverable only. Use -DCHOIR_BUILD_PLUGIN=OFF for DSP tests.")
 endif()
 if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
  message(FATAL_ERROR "Use the Visual Studio x64 generator (-A x64).")
 endif()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
 FetchContent_MakeAvailable(JUCE)
 juce_add_plugin(NooriVoiceChoir COMPANY_NAME "Noori Turab" BUNDLE_ID "audio.nooriturab.voicechoir" PLUGIN_MANUFACTURER_CODE Ntur PLUGIN_CODE Nvch PRODUCT_NAME "NOORI CHOIR" FORMATS VST3 IS_ARA_EFFECT FALSE IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Generator)
 juce_add_binary_data(NooriBrand NAMESPACE NooriBrand SOURCES Assets/NOORI.svg)
 target_include_directories(NooriVoiceChoir PRIVATE ThirdParty "${CMAKE_CURRENT_BINARY_DIR}/generated")
 target_sources(NooriVoiceChoir PRIVATE Source/Plugin.cpp Source/Editor.cpp Source/Licensing.cpp)
 target_compile_definitions(NooriVoiceChoir PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_compile_options(NooriVoiceChoir PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 target_link_libraries(NooriVoiceChoir PRIVATE NooriBrand juce::juce_audio_utils juce::juce_cryptography bcrypt crypt32 advapi32 PUBLIC juce::juce_recommended_config_flags)
 add_executable(choir_plugin_tests tests/PluginTests.cpp)
 target_include_directories(choir_plugin_tests PRIVATE Source ThirdParty "${juce_SOURCE_DIR}/modules" $<TARGET_PROPERTY:NooriVoiceChoir,INCLUDE_DIRECTORIES>)
 target_compile_definitions(choir_plugin_tests PRIVATE $<TARGET_PROPERTY:NooriVoiceChoir,COMPILE_DEFINITIONS>)
 target_link_libraries(choir_plugin_tests PRIVATE NooriVoiceChoir)
 add_executable(choir_file_tests tests/FileTests.cpp)
 target_include_directories(choir_file_tests PRIVATE Source ThirdParty "${juce_SOURCE_DIR}/modules" $<TARGET_PROPERTY:NooriVoiceChoir,INCLUDE_DIRECTORIES>)
 target_compile_definitions(choir_file_tests PRIVATE $<TARGET_PROPERTY:NooriVoiceChoir,COMPILE_DEFINITIONS>)
 target_link_libraries(choir_file_tests PRIVATE NooriVoiceChoir)
 add_executable(choir_licensing_tests tests/LicensingTests.cpp)
 target_include_directories(choir_licensing_tests PRIVATE Source ThirdParty "${juce_SOURCE_DIR}/modules" $<TARGET_PROPERTY:NooriVoiceChoir,INCLUDE_DIRECTORIES>)
 target_compile_definitions(choir_licensing_tests PRIVATE $<TARGET_PROPERTY:NooriVoiceChoir,COMPILE_DEFINITIONS>)
 target_link_libraries(choir_licensing_tests PRIVATE NooriVoiceChoir)
 add_test(NAME ChoirOnlineLicensing COMMAND choir_licensing_tests "${CMAKE_BINARY_DIR}/license-fixture.json")
 set_tests_properties(ChoirOnlineLicensing PROPERTIES TIMEOUT 120)
 add_test(NAME ChoirWavImportExport COMMAND choir_file_tests "${CMAKE_BINARY_DIR}")
 set_tests_properties(ChoirWavImportExport PROPERTIES TIMEOUT 300)
 add_test(NAME ChoirProcessorAndEditor COMMAND choir_plugin_tests "${CMAKE_BINARY_DIR}")
 set_tests_properties(ChoirProcessorAndEditor PROPERTIES TIMEOUT 300)
endif()
