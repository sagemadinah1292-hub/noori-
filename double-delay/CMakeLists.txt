cmake_minimum_required(VERSION 3.24)
project(NooriDoubleDelay VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DELAY_BUILD_PLUGIN "Build VST3 and processor tests" ON)
include(CTest)
add_executable(delay_engine_tests tests/EngineTests.cpp)
target_include_directories(delay_engine_tests PRIVATE Source)
add_test(NAME DelayEngine COMMAND delay_engine_tests)
if(DELAY_BUILD_PLUGIN)
 if(NOT WIN32)
  message(FATAL_ERROR "Windows x64 deliverable only. Use -DDELAY_BUILD_PLUGIN=OFF for DSP tests.")
 endif()
 include(FetchContent)
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
 FetchContent_MakeAvailable(JUCE)
 juce_add_plugin(NooriDoubleDelay COMPANY_NAME "Noori Turab" BUNDLE_ID "audio.nooriturab.doubledelay" PLUGIN_MANUFACTURER_CODE Ntur PLUGIN_CODE Nddl PRODUCT_NAME "NOORI Double Delay" FORMATS VST3 IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE EDITOR_WANTS_KEYBOARD_FOCUS FALSE COPY_PLUGIN_AFTER_BUILD FALSE VST3_CATEGORIES Fx Delay)
 target_sources(NooriDoubleDelay PRIVATE Source/Plugin.cpp Source/Editor.cpp)
 target_compile_definitions(NooriDoubleDelay PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
 target_compile_options(NooriDoubleDelay PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 target_link_libraries(NooriDoubleDelay PRIVATE juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
 add_executable(delay_plugin_tests tests/PluginTests.cpp)
 target_include_directories(delay_plugin_tests PRIVATE Source "${juce_SOURCE_DIR}/modules" $<TARGET_PROPERTY:NooriDoubleDelay,INCLUDE_DIRECTORIES>)
 target_compile_definitions(delay_plugin_tests PRIVATE $<TARGET_PROPERTY:NooriDoubleDelay,COMPILE_DEFINITIONS>)
 target_link_libraries(delay_plugin_tests PRIVATE NooriDoubleDelay)
 add_test(NAME DelayProcessorAndEditor COMMAND delay_plugin_tests "${CMAKE_BINARY_DIR}")
 set_tests_properties(DelayProcessorAndEditor PROPERTIES TIMEOUT 120)
endif()
