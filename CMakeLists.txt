cmake_minimum_required(VERSION 3.24)
project(NooriTurabTempo VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NOORI_BUILD_PLUGIN "Build the Windows VST3 (other systems: compile-check only)" ON)
option(NOORI_COMPILE_CHECK "Allow compiling shared plugin code on Linux, without plugin deliverables" OFF)
option(NOORI_ENABLE_ARA "Build official ARA extension" ON)
include(FetchContent)
FetchContent_Declare(signalsmith-linear GIT_REPOSITORY https://github.com/Signalsmith-Audio/linear.git GIT_TAG 0.6.4 GIT_SHALLOW TRUE)
FetchContent_Declare(signalsmith-stretch GIT_REPOSITORY https://github.com/Signalsmith-Audio/signalsmith-stretch.git GIT_TAG a670068d9aeb64913331d5cc29337b19a457a7df)
FetchContent_MakeAvailable(signalsmith-linear signalsmith-stretch)
add_library(noori_dsp Source/Engine.cpp)
target_include_directories(noori_dsp PUBLIC Source)
target_link_libraries(noori_dsp PRIVATE signalsmith-stretch)
set_target_properties(noori_dsp PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
 target_compile_options(noori_dsp PRIVATE /W4 /permissive- /utf-8)
else()
 target_compile_options(noori_dsp PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
 add_executable(noori_tests tests/EngineTests.cpp)
 target_link_libraries(noori_tests PRIVATE noori_dsp)
 add_test(NAME DSP COMMAND noori_tests)
endif()
if(NOORI_BUILD_PLUGIN)
 if(NOT WIN32 AND NOT NOORI_COMPILE_CHECK)
  message(FATAL_ERROR "Only Windows VST3 delivery is supported. Use -DNOORI_BUILD_PLUGIN=OFF for DSP tests.")
 endif()
 FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 29396c22c93392d6738e021b83196283d6e4d850)
 FetchContent_MakeAvailable(JUCE)
 if(NOORI_ENABLE_ARA)
  FetchContent_Declare(ARA_SDK GIT_REPOSITORY https://github.com/Celemony/ARA_SDK.git GIT_TAG e723ce48a917b1ce2d7851c78e468f36b51ea151 GIT_SUBMODULES ARA_API ARA_Library)
  FetchContent_MakeAvailable(ARA_SDK)
  juce_set_ara_sdk_path(${ara_sdk_SOURCE_DIR})
 endif()
 set(NOORI_FORMATS "")
 if(WIN32)
  set(NOORI_FORMATS VST3)
 endif()
 juce_add_plugin(NooriTurabTempo
  COMPANY_NAME "Noori Turab" COMPANY_WEBSITE "https://github.com/sagemadinah1292-hub/noori-" BUNDLE_ID "audio.nooriturab.tempo" PLUGIN_MANUFACTURER_CODE Ntur
  PLUGIN_CODE Ntmp PRODUCT_NAME "NooriTurabTempo" FORMATS ${NOORI_FORMATS}
  IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE
  EDITOR_WANTS_KEYBOARD_FOCUS TRUE
  COPY_PLUGIN_AFTER_BUILD FALSE IS_ARA_EFFECT ${NOORI_ENABLE_ARA}
  ARA_FACTORY_ID "\"audio.nooriturab.tempo.factory.0.2.0\""
  ARA_DOCUMENT_ARCHIVE_ID "\"audio.nooriturab.tempo.archive.3\""
  ARA_COMPATIBLE_ARCHIVE_IDS "\"audio.nooriturab.tempo.archive.1\\naudio.nooriturab.tempo.archive.2\""
  VST3_CATEGORIES Fx Tools)
 target_compile_options(NooriTurabTempo PRIVATE $<$<CXX_COMPILER_ID:MSVC>:/utf-8>)
 target_sources(NooriTurabTempo PRIVATE Source/PluginProcessor.cpp Source/PluginEditor.cpp Source/ARAIntegration.cpp)
 target_link_libraries(NooriTurabTempo PRIVATE noori_dsp juce::juce_audio_utils PUBLIC juce::juce_recommended_config_flags)
 target_compile_definitions(NooriTurabTempo PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0 JUCE_VST3_CAN_REPLACE_VST2=0)
endif()

if(NOORI_BUILD_PLUGIN AND BUILD_TESTING)
 add_executable(noori_plugin_smoke tests/PluginSmoke.cpp)
 target_include_directories(noori_plugin_smoke PRIVATE Source "${juce_SOURCE_DIR}/modules" "${ara_sdk_SOURCE_DIR}")
 target_link_libraries(noori_plugin_smoke PRIVATE NooriTurabTempo)
 if(NOORI_ENABLE_ARA)
  target_sources(noori_plugin_smoke PRIVATE tests/ARAImplementation.cpp)
 endif()
 target_include_directories(noori_plugin_smoke PRIVATE $<TARGET_PROPERTY:NooriTurabTempo,INCLUDE_DIRECTORIES>)
 target_compile_definitions(noori_plugin_smoke PRIVATE $<TARGET_PROPERTY:NooriTurabTempo,COMPILE_DEFINITIONS>)
 add_test(NAME ProcessorAndEditor COMMAND noori_plugin_smoke "${CMAKE_BINARY_DIR}")
 set_tests_properties(ProcessorAndEditor PROPERTIES TIMEOUT 120)
endif()
